# scripts/env-scan-ignore.txt — UPPER_CASE names scripts/check-env-example.sh finds but that don't belong
# in .env.example. One per line: "NAME # reason". A name here must not also be in .env.example.

# Set by the OS or the shell.
HOME # the user's home directory (OS)
PATH # executable search path (OS)
USER # login name (OS)
EUID # the effective user id (bash sets it)
TMPDIR # temp directory (OS)
COLUMNS # terminal width (shell); the statusline falls back to it
PROMPT_COMMAND # bash prompt hook; the Claude installer appends to it
XDG_RUNTIME_DIR # per-user runtime dir (OS); sindri doctor reports it and masks it in the graphify sandbox
DISPLAY # the X11 session sets it; --init-launcher uses it to tell a GUI session from a headless box
TERM_PROGRAM # the terminal sets it; aw-meta-launch --init-launcher picks the viewer by it
WAYLAND_DISPLAY # the Wayland session sets it; --init-launcher uses it to tell a GUI session from a headless box

# Owned by another tool; the code only reports or forwards it.
NODE_OPTIONS # Node's own option variable; sindri doctor checks it for --use-env-proxy
NODE_USE_ENV_PROXY # Node's own proxy switch; sindri doctor reports it
GIT_CONFIG_GLOBAL # git's own variable; sindri onboard forwards it to the git it runs

# Set by the host (Claude Code, Cursor) when it runs our hooks or installers.
CLAUDECODE # set by Claude Code inside its sessions; the installer skips shell edits there, and sindri's CLI classes itself as a session
CLAUDE_CODE_ENTRYPOINT # set by Claude Code inside its sessions; a pinned probe marker (sindri's CLI classes itself as a session)
CLAUDE_CODE_EXECPATH # set by Claude Code inside its sessions; a pinned probe marker (sindri's CLI classes itself as a session)
CLAUDE_CODE_SESSION_ID # set by Claude Code inside its sessions; a pinned probe marker (sindri's CLI classes itself as a session)
CLAUDE_PID # set by Claude Code inside its sessions; a pinned probe marker (sindri's CLI classes itself as a session)
CODEX_SANDBOX # set by Codex in its sandboxed sessions; sindri's CLI reads it to class itself as a session
CURSOR_AGENT # set by Cursor's agent in its sessions (unverified); sindri's CLI reads it to class itself as a session
CURSOR_PROJECT_DIR # set by Cursor for hook commands
CURSOR_TRANSCRIPT_PATH # set by Cursor for hook commands

# Internal plumbing: set by the toolkit's own scripts for the scripts they call, not by users.
AW_PROVIDER # set by the hook adapters and installers (claude, codex, cursor)
AW_ADAPTER_EVENT # set by the hook adapters' argument parser
AW_FAILOPEN_INPUT_LOGGED # set by the adapters after they log a bad hook input once
AW_HOOK_ADAPTER_FLAGS # passed by the hook installer to the adapter command line it writes
AW_DRY_RUN # set by setup.sh --dry-run for the installers it calls
AW_MCP_SERVERS # set by setup.sh (the MCP server list) for the provider installers
AW_MCP_RETIRED # set by setup.sh (servers the toolkit no longer ships) for the provider installers to remove
AW_EXTERNAL_SKILL_DIRS # set by setup.sh (fetched skill packs) for the provider installers
WITH_SINDRI # set by setup.sh --with-sindri for the Claude installer
AW_JUDGE_CHILD # recursion guard the judge sets on the agent CLIs it starts
AW_SINDRI_CHILD # recursion guard sindri sets on the processes it starts; sindri's CLI classes itself as a session under it
AW_SINDRI_PROVER # set by scripts/sindri-prove.sh human; labels sindri's CLI calls 'prover'
LOCK_NAME # a parameter callers set before sourcing skills/_shared/skill-lock.sh
REPO # sync-rules.sh's positional argument; the script resets it before reading arguments, so the environment can't set it

# Script-local UPPER_CASE variables the scan can't tell from overrides (config/statusline.sh).
BAR_FILL # statusline local (context bar)
BAR_FILL5 # statusline local (5-hour bar)
COST # statusline local (session cost)
CTX_PCT # statusline local (context percent)
LINES_ADD # statusline local (lines added)
LINES_DEL # statusline local (lines removed)
TOTAL_MIN # statusline local (session minutes)

# Examples in a doc comment, not reads (scripts/deps/scan.mjs).
X # doc-comment example of an env fallback
X_BIN # doc-comment example of a binary override

# Set by CI or by a caller, never by hand.
GITHUB_STEP_SUMMARY # GitHub Actions sets it; scripts/ci/lane-budgets.mjs appends its table there
AW_HEAVY_LOCK_HELD # set to 1 by a caller that already holds a heavy-job lock slot (bash-tests.sh, the scaffold e2e)
LAUNCHER_KEEP_EMPTY_SEED # a shell variable meta-lane.sh sets for launcher_expand (1: an existing but empty prompt file still passes ""); not user configuration
TMUX # set by tmux inside a client; scripts/test-tmux.sh unsets it so tests can never reach the box's own server
TMUX_TMPDIR # tmux's socket directory; tests point it at a temp dir (scripts/test-tmux.sh)
NUGET_PACKAGES # NuGet's own variable; scripts/ci/bash-tests.sh keeps the real package cache reachable from a suite's temp HOME
PPID # the shell sets it (the aw:fleet hook walks up from it)
EPOCHREALTIME # the shell sets it (bash 5; the aw:fleet hook timestamps with it)
AW_HOOK_PROVIDER # set by the Codex and Cursor adapters (config/hooks/adapters/common.sh) for the hook they run, never by the user; the aw:fleet hook maps it to the provider directory
EDITOR # the user's own editor choice; sindri-pod.sh profile edit opens the profile in it (vi when unset)
AW_SINDRI_ONYX_URL # set by the chart in the pod, read there by sindri-pod.sh (connect --onyx) to find Onyx's port
